add_executable(goovm-llc goovm-llc.cpp gir_export.cpp)
target_include_directories(goovm-llc SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(GOOVM_LLVM_DEFS NATIVE_COMMAND ${LLVM_DEFINITIONS})
target_compile_definitions(goovm-llc PRIVATE ${GOOVM_LLVM_DEFS})
target_compile_options(goovm-llc PRIVATE -Wall -Wextra -Wno-unused-parameter -fno-exceptions)
if (NOT LLVM_ENABLE_RTTI)
    target_compile_options(goovm-llc PRIVATE -fno-rtti)
endif ()

if (LLVM_LINK_LLVM_DYLIB)
    target_link_libraries(goovm-llc PRIVATE LLVM)
else ()
    llvm_map_components_to_libnames(GOOVM_LLVM_LIBS core bitreader support target
        x86codegen x86asmparser x86desc x86info)
    target_link_libraries(goovm-llc PRIVATE ${GOOVM_LLVM_LIBS})
endif ()
target_link_libraries(goovm-llc PRIVATE goovm_backend_cpp)

if (GOOVM_WITH_GOOSE)
    if (NOT EXISTS "${GOOVM_GOOSE_C}")
        message(FATAL_ERROR "GOOVM_WITH_GOOSE: ${GOOVM_GOOSE_C} not found; generate it with\n"
                "  goose --standalone --header cg.h -o cg.c backend-goose/cg.goose\n"
                "and pass -DGOOVM_GOOSE_C=/path/to/cg.c")
    endif ()
    set(goose_h "${GOOVM_GOOSE_H}")
    if (NOT goose_h)
        get_filename_component(goose_dir "${GOOVM_GOOSE_C}" DIRECTORY)
        set(goose_h "${goose_dir}/cg.h")
    endif ()
    if (NOT EXISTS "${goose_h}")
        message(FATAL_ERROR "GOOVM_WITH_GOOSE: header ${goose_h} not found (set GOOVM_GOOSE_H)")
    endif ()
    # The generated C is compiled like any C file; warnings are its own.
    add_library(goovm_backend_goose STATIC "${GOOVM_GOOSE_C}")
    set_source_files_properties("${GOOVM_GOOSE_C}" PROPERTIES LANGUAGE C)
    target_compile_options(goovm_backend_goose PRIVATE -w)
    target_compile_definitions(goovm-llc PRIVATE GOOVM_WITH_GOOSE "GOOVM_GOOSE_HEADER=\"${goose_h}\"")
    target_link_libraries(goovm-llc PRIVATE goovm_backend_goose m pthread)
endif ()
